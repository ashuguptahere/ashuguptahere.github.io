% Tighten the space the class leaves under the name (it defaults to \medskip).
\def\nameskip{\smallskip}

% Never break a line at an explicit hyphen. A break inside "RF-DETR" or
% "edge-AI" makes PDF text extraction emit "RFDETR"/"edgeAI", so the ATS never
% matches the keyword. (Automatic hyphenation is fine: those are discretionary
% breaks and extraction drops them cleanly.)
\exhyphenpenalty=10000

% With hyphen breaks forbidden, a few long comma lists have no break near the
% margin. Let TeX loosen the spacing slightly in a final pass instead of
% letting a line run past the margin.
\emergencystretch=2em

% needspace.sty is not in every TeX Live install, so reserve the space inline:
% skip down by #1 with breaking forbidden, then take the skip back. If those
% lines would not fit, the page breaks at the -100 penalty instead, which keeps
% a section heading from stranding itself at the foot of a page.
\newcommand{\reserveSpace}[1]{%
  \par\penalty-100%
  \begingroup
    \dimen0=#1\relax
    \vskip\dimen0
    \penalty10000%
    \vskip-\dimen0
  \endgroup}

-- Minimal YAML loader (supports the subset used by data.yaml).
-- If lyaml is available, we will prefer it.

function trim(s)
  return (s:gsub('^%s+', ''):gsub('%s+$', ''))
end

function parse_quoted_scalar(v)
  v = trim(v)
  if v == '' then return '' end

  local first = v:sub(1,1)
  local last  = v:sub(-1)

  if first == "'" and last == "'" then
    local inner = v:sub(2, -2)
    inner = inner:gsub("''", "'")
    return inner
  end

  if first == '"' and last == '"' then
    local inner = v:sub(2, -2)
    inner = inner:gsub('\\\\', '\\')
    inner = inner:gsub('\\"', '"')
    inner = inner:gsub('\\n', '\n')
    inner = inner:gsub('\\t', '\t')
    return inner
  end

  -- unquoted
  return v
end

function is_blank_or_comment(line)
  if not line then return true end
  local t = trim(line)
  return t == '' or t:sub(1,1) == '#'
end

function count_indent(line)
  local _, n = line:find('^%s*')
  return n or 0
end

function yaml_load_minimal(path)
  local f = io.open(path, 'r')
  if not f then
    error('Could not open YAML file: ' .. path)
  end
  local lines = {}
  for line in f:lines() do
    table.insert(lines, line)
  end
  f:close()

  local root = {}
  local stack = { {indent = -1, node = root} }

  local function current()
    return stack[#stack]
  end

  local function pop_to(indent)
    while #stack > 1 and current().indent >= indent do
      table.remove(stack)
    end
  end

  for i=1, #lines do
    local line = lines[i]
    if is_blank_or_comment(line) then
      goto continue
    end

    local indent = count_indent(line)
    local content = trim(line)

    pop_to(indent)
    local parent = current().node

    -- List item
    if content:sub(1,1) == '-' then
      local rest = trim(content:sub(2))

      if rest == '' then
        local item = {}
        table.insert(parent, item)
        table.insert(stack, {indent = indent, node = item})
        goto continue
      end

      local k, v = rest:match('^([%w_%-]+)%s*:%s*(.*)$')
      if k then
        local item = {}
        item[k] = (v == '' and {} or parse_quoted_scalar(v))
        table.insert(parent, item)
        if v == '' then
          table.insert(stack, {indent = indent, node = item[k]})
        else
          table.insert(stack, {indent = indent, node = item})
        end
      else
        table.insert(parent, parse_quoted_scalar(rest))
      end

      goto continue
    end

    -- Key/value
    local key, val = content:match('^([^:]+):%s*(.*)$')
    if not key then
      error('Could not parse YAML at line ' .. i .. ': ' .. content)
    end

    key = trim(key)

    if val == '' then
      parent[key] = {}
      table.insert(stack, {indent = indent, node = parent[key]})
    else
      parent[key] = parse_quoted_scalar(val)
    end

    ::continue::
  end

  return root
end

function load_resume_yaml(path)
  local ok, lyaml = pcall(require, 'lyaml')
  if ok and lyaml then
    local f = io.open(path, 'r')
    if not f then error('Could not open YAML file: ' .. path) end
    local s = f:read('*all')
    f:close()
    return lyaml.load(s)
  end
  return yaml_load_minimal(path)
end

-- LaTeX escaping for plain text fields (NOT for fields that already contain LaTeX).
function latex_escape(s)
  if s == nil then return '' end
  s = tostring(s)
  s = s:gsub('\\', '\\textbackslash{}')
  s = s:gsub('{', '\\{')
  s = s:gsub('}', '\\}')
  s = s:gsub('%%', '\\%')
  s = s:gsub('%$', '\\$')
  s = s:gsub('#', '\\#')
  s = s:gsub('&', '\\&')
  s = s:gsub('_', '\\_')
  s = s:gsub('~', '\\textasciitilde{}')
  s = s:gsub('%^', '\\textasciicircum{}')
  return s
end

-- Safer for URLs in \href: detokenize prevents TeX from treating special chars.
function href(url, text)
  url = url or ''
  text = text or ''
  return '\\href{\\detokenize{' .. url .. '}}{' .. text .. '}'
end

resume = load_resume_yaml('data.yaml')

function safe_ipairs(t)
  if type(t) ~= 'table' then
    return function() end
  end
  return ipairs(t)
end

-- Contact separator. \textperiodcentered survives PDF text extraction as U+00B7,
-- unlike the class's $\diamond$, which extracts as nothing at all. The spaces
-- must be real glue, not \, kerns, or the contact line has no legal breakpoint
-- and runs off the page instead of wrapping.
CONTACT_SEP = ' \\textperiodcentered\\ '

-- The class prints addresses through \printaddress, which wraps each line in an
-- unbreakable \centerline and caps out at three lines. Full profile URLs blow
-- past both limits, so bypass \address entirely and emit a real centred
-- paragraph, which wraps on its own.
function render_header()
  local b = resume.basics or {}

  -- A center environment would add \topsep on top of the class's \nameskip and
  -- the parskip package's inter-paragraph glue, leaving a visible hole under
  -- the name. A bare \centering group adds none of that.
  tex.print('\\vspace{-\\parskip}')
  tex.print('{\\centering')

  if b.titles and b.titles ~= '' then
    tex.print('{\\bf ' .. latex_escape(b.titles) .. '}\\\\[2pt]')
  end

  local where = {}
  if b.location and b.location ~= '' then
    table.insert(where, latex_escape(b.location))
  end
  if b.work_authorization and b.work_authorization ~= '' then
    table.insert(where, latex_escape(b.work_authorization))
  end
  if #where > 0 then
    tex.print(table.concat(where, CONTACT_SEP) .. '\\\\')
  end

  -- Show the URL itself as the link text: ATS parsers read the text layer and
  -- almost never follow PDF link annotations, so "LinkedIn" leaves them with
  -- no URL at all.
  local contacts = {}
  table.insert(contacts, href('mailto:' .. (b.email or ''), latex_escape(b.email or '')))
  for _, link in safe_ipairs(b.links) do
    local shown = link.display
    if not shown or shown == '' then shown = link.label or '' end
    table.insert(contacts, href(link.url or '', latex_escape(shown)))
  end
  tex.print(table.concat(contacts, CONTACT_SEP))

  tex.print('\\par}')
  tex.print('\\vspace{-6pt}')
end

-- Keep a section heading with at least its first lines, so a heading never
-- strands itself at the foot of a page.
-- PDF document properties. Some ATS read the XMP/Info metadata in addition to
-- the page text, so mirror the visible content into it.
function render_pdf_metadata()
  local b = resume.basics or {}

  local keywords = {}
  for _, s in safe_ipairs(resume.skills) do
    for _, v in safe_ipairs(s.items) do
      table.insert(keywords, latex_escape(v))
    end
  end

  local subject = b.headline or ''
  if b.titles and b.titles ~= '' then
    subject = subject .. ' - ' .. b.titles
  end

  tex.print('\\hypersetup{')
  tex.print('  pdftitle={' .. latex_escape((b.name or '') .. ' - ' .. (b.headline or '')) .. '},')
  tex.print('  pdfauthor={' .. latex_escape(b.name or '') .. '},')
  tex.print('  pdfsubject={' .. latex_escape(subject) .. '},')
  tex.print('  pdfkeywords={' .. table.concat(keywords, ', ') .. '}')
  tex.print('}')
end

-- Name and "Page N of M" in the footer of every page, so a page separated from
-- the rest is still identifiable. M comes from \label{lastpage} at the end of
-- the document, so this needs two LaTeX passes to settle.
function render_page_footer()
  local b = resume.basics or {}
  tex.print('\\makeatletter')
  tex.print('\\def\\ps@resumefooter{%')
  tex.print('  \\def\\@oddfoot{\\small\\textit{' .. latex_escape(b.name or '') .. '}\\hfil\\small\\textit{Page \\thepage\\ of \\pageref*{lastpage}}}%')
  tex.print('  \\let\\@evenfoot\\@oddfoot')
  tex.print('  \\def\\@oddhead{}\\let\\@evenhead\\@oddhead}')
  tex.print('\\makeatother')
  tex.print('\\pagestyle{resumefooter}')
end

function begin_rsection(title)
  tex.print('\\reserveSpace{4\\baselineskip}')
  tex.print('\\begin{rSection}{' .. latex_escape(title) .. '}')
end

function end_rsection()
  tex.print('\\end{rSection}')
end

function print_itemize(items_latex, itemsep)
  tex.print('\\begin{itemize}')
  if itemsep then
    tex.print('\\setlength{\\itemsep}{' .. itemsep .. '}')
  else
    tex.print('\\itemsep -6pt {}')
  end
  for _, it in safe_ipairs(items_latex) do
    tex.print('\\item ' .. it)
  end
  tex.print('\\end{itemize}')
end

function render_summary()
  local text = (resume.summary or {}).text
  if not text or text == '' then return end

  begin_rsection('SUMMARY')
  tex.print(latex_escape(text))
  end_rsection()
end

-- Harvard entry format: organisation (bold) with location right-aligned, then
-- title (italic) with dates right-aligned.
function render_experience()
  begin_rsection('EXPERIENCE')

  for idx, job in safe_ipairs(resume.experience) do
    tex.print('\\textbf{' .. latex_escape(job.company or '') .. '} \\hfill ' .. latex_escape(job.location or '') .. '\\\\')

    local role = latex_escape(job.role or '')
    if job.role_url and job.role_url ~= '' then
      role = href(job.role_url, role)
    end
    tex.print('\\textit{' .. role .. '} \\hfill ' .. latex_escape(job.dates or ''))

    if type(job.stack) == 'table' and #job.stack > 0 then
      local items = {}
      for _, v in safe_ipairs(job.stack) do
        table.insert(items, latex_escape(v))
      end
      tex.print('\\\\\\textbf{Stack:} ' .. table.concat(items, ', '))
    end

    local vs = job.vspace_after_header
    if vs and vs ~= '0pt' and vs ~= '' then
      tex.print('\\vspace{' .. vs .. '}')
    end

    print_itemize(job.bullets_latex)

    if idx < #resume.experience then
      tex.print('')
    end
  end

  end_rsection()
end

-- Plain wrapped paragraphs, not a tabular: PDF text extraction splits table
-- cells from their row labels (the old "Models" row lost its values entirely).
function render_skills()
  begin_rsection('SKILLS & INTERESTS')

  -- One bullet per category (the top layer only); the skills themselves stay
  -- comma-separated inside the bullet.
  tex.print('\\begin{itemize}')
  tex.print('\\setlength{\\itemsep}{-6pt}')

  for _, s in safe_ipairs(resume.skills) do
    local items = {}
    for _, v in safe_ipairs(s.items) do
      table.insert(items, latex_escape(v))
    end
    tex.print('\\item \\textbf{' .. latex_escape(s.category or '') .. ':} ' .. table.concat(items, ', '))
  end

  tex.print('\\end{itemize}')
  end_rsection()
end

function render_projects()
  begin_rsection('PROJECTS')
  tex.print('\\begin{itemize}')
  tex.print('\\setlength{\\itemsep}{-5pt}')
  for _, p in safe_ipairs(resume.projects) do
    local name = latex_escape(p.name or '')
    local url = p.url or ''
    local tail = p.tail_latex or ''
    -- Not every project has a public repo; an empty \href would render as a
    -- dead link.
    if url ~= '' then
      name = href(url, name)
    end
    tex.print('\\item ' .. name .. tail)
  end
  tex.print('\\end{itemize}')
  end_rsection()
end

function render_leadership()
  local items = (resume.leadership or {}).items_latex
  if type(items) ~= 'table' or #items == 0 then return end

  begin_rsection('Leadership & Activities')
  tex.print('\\begin{itemize}')
  tex.print('\\setlength{\\itemsep}{-6pt}')
  for _, line in safe_ipairs(items) do
    tex.print('\\item ' .. line)
  end
  tex.print('\\end{itemize}')
  end_rsection()
end

function render_achievements()
  begin_rsection('Achievements')
  tex.print('\\begin{itemize}')
  tex.print('\\setlength{\\itemsep}{-6pt}')
  for _, line in safe_ipairs((resume.achievements or {}).items_latex) do
    tex.print('\\item ' .. line)
  end
  tex.print('\\end{itemize}')
  end_rsection()
end

function render_education()
  begin_rsection('Education')
  local line = ((resume.education or {}).line_latex) or ''
  tex.print(line)
  end_rsection()
end

-- Single column: multicols reading order is fragile under PDF text extraction
-- and buys nothing here.
function render_certifications()
  begin_rsection('CERTIFICATIONS')
  tex.print('\\begin{itemize}')
  tex.print('\\setlength{\\itemsep}{-6pt}')

  for _, c in safe_ipairs((resume.certifications or {}).items) do
    local name = latex_escape(c.name or '')
    local url = c.url or ''
    tex.print('\\item ' .. href(url, name))
  end

  tex.print('\\end{itemize}')
  end_rsection()
end

function render_resume()
  render_page_footer()
  render_header()
  render_summary()
  render_experience()
  render_education()
  render_skills()
  render_projects()
  render_achievements()
  render_leadership()
  render_certifications()
end
